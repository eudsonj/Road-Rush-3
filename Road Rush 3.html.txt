  <script>
    /* =========================================================================
       WEB AUDIO SYNTHESIZER (Car Engine 3 - Deep Biturbo Sport Engine FX)
       ========================================================================= */
    class SoundEngine {
      constructor() {
        this.ctx = null;
        this.subRumbleOsc = null;
        this.cylOsc1 = null;
        this.cylOsc2 = null;
        this.cylOsc3 = null;

        this.exhaustNoiseNode = null;
        this.exhaustFilter = null;
        this.exhaustGain = null;

        this.engineGain = null;
        this.engineFilter = null;
        this.distortionNode = null;

        this.turboOsc1 = null;
        this.turboOsc2 = null;
        this.turboGain = null;
        this.turboFilter = null;

        this.isInitialized = false;
        this.wasAccelerating = false;
        this.lastPopTime = 0;
        this.isEngineRunning = false;
      }

      init() {
        if (this.isInitialized) {
          if (this.ctx && this.ctx.state === 'suspended') {
            this.ctx.resume();
          }
          return;
        }
        try {
          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          this.ctx = new AudioCtx();
          if (this.ctx.state === 'suspended') {
            this.ctx.resume();
          }
          const now = this.ctx.currentTime;

          this.subRumbleOsc = this.ctx.createOscillator();
          this.subRumbleOsc.type = 'sine';
          this.subRumbleOsc.frequency.setValueAtTime(20, now);

          this.cylOsc1 = this.ctx.createOscillator();
          this.cylOsc1.type = 'sawtooth';
          this.cylOsc1.frequency.setValueAtTime(28, now);

          this.cylOsc2 = this.ctx.createOscillator();
          this.cylOsc2.type = 'triangle';
          this.cylOsc2.frequency.setValueAtTime(42, now);

          this.cylOsc3 = this.ctx.createOscillator();
          this.cylOsc3.type = 'sawtooth';
          this.cylOsc3.frequency.setValueAtTime(56, now);

          this.distortionNode = this.ctx.createWaveShaper();
          const curve = new Float32Array(256);
          for (let i = 0; i < 256; i++) {
            let x = (i * 2) / 256 - 1;
            curve[i] = (Math.PI + 3) * x / (Math.PI + 3 * Math.abs(x));
          }
          this.distortionNode.curve = curve;

          this.engineFilter = this.ctx.createBiquadFilter();
          this.engineFilter.type = 'lowpass';
          this.engineFilter.frequency.setValueAtTime(200, now);
          this.engineFilter.Q.setValueAtTime(4.5, now);

          this.engineGain = this.ctx.createGain();
          this.engineGain.gain.setValueAtTime(0.18, now);

          this.cylOsc1.connect(this.distortionNode);
          this.cylOsc2.connect(this.distortionNode);
          this.cylOsc3.connect(this.distortionNode);
          this.subRumbleOsc.connect(this.engineFilter);

          this.distortionNode.connect(this.engineFilter);
          this.engineFilter.connect(this.engineGain);
          this.engineGain.connect(this.ctx.destination);

          this.cylOsc1.start(now);
          this.cylOsc2.start(now);
          this.cylOsc3.start(now);
          this.subRumbleOsc.start(now);

          const bufferSize = this.ctx.sampleRate * 2;
          const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
          const noiseData = noiseBuffer.getChannelData(0);
          for (let i = 0; i < bufferSize; i++) {
            noiseData[i] = Math.random() * 2 - 1;
          }

          this.exhaustNoiseNode = this.ctx.createBufferSource();
          this.exhaustNoiseNode.buffer = noiseBuffer;
          this.exhaustNoiseNode.loop = true;

          this.exhaustFilter = this.ctx.createBiquadFilter();
          this.exhaustFilter.type = 'bandpass';
          this.exhaustFilter.frequency.setValueAtTime(450, now);
          this.exhaustFilter.Q.setValueAtTime(2.8, now);

          this.exhaustGain = this.ctx.createGain();
          this.exhaustGain.gain.setValueAtTime(0.02, now);

          this.exhaustNoiseNode.connect(this.exhaustFilter);
          this.exhaustFilter.connect(this.exhaustGain);
          this.exhaustGain.connect(this.ctx.destination);
          this.exhaustNoiseNode.start(now);

          this.turboOsc1 = this.ctx.createOscillator();
          this.turboOsc1.type = 'sine';
          this.turboOsc1.frequency.setValueAtTime(1100, now);

          this.turboOsc2 = this.ctx.createOscillator();
          this.turboOsc2.type = 'sine';
          this.turboOsc2.frequency.setValueAtTime(1135, now);

          this.turboFilter = this.ctx.createBiquadFilter();
          this.turboFilter.type = 'bandpass';
          this.turboFilter.frequency.setValueAtTime(1600, now);
          this.turboFilter.Q.setValueAtTime(7.0, now);

          this.turboGain = this.ctx.createGain();
          this.turboGain.gain.setValueAtTime(0.005, now);

          this.turboOsc1.connect(this.turboFilter);
          this.turboOsc2.connect(this.turboFilter);
          this.turboFilter.connect(this.turboGain);
          this.turboGain.connect(this.ctx.destination);

          this.turboOsc1.start(now);
          this.turboOsc2.start(now);

          this.isInitialized = true;
          this.isEngineRunning = true;
        } catch (e) {
          console.warn("Audio Context init error", e);
        }
      }

      startEngine() {
        this.isEngineRunning = true;
      }

      stopEngine() {
        this.isEngineRunning = false;
        if (!this.isInitialized || !this.ctx) return;
        const now = this.ctx.currentTime;
        [this.engineGain, this.exhaustGain, this.turboGain].forEach(g => {
          if (g && g.gain) {
            g.gain.cancelScheduledValues(now);
            g.gain.setValueAtTime(0, now);
          }
        });
      }

      playExhaustPop() {
        if (!this.isInitialized || !this.ctx || !this.isEngineRunning) return;
        const now = this.ctx.currentTime;
        if (now - this.lastPopTime < 0.22) return;
        this.lastPopTime = now;

        const popCount = 1 + Math.floor(Math.random() * 3);
        for (let p = 0; p < popCount; p++) {
          const delay = p * (0.05 + Math.random() * 0.08);
          const popTime = now + delay;

          const popOsc = this.ctx.createOscillator();
          popOsc.type = 'triangle';
          popOsc.frequency.setValueAtTime(120 + Math.random() * 50, popTime);
          popOsc.frequency.exponentialRampToValueAtTime(20, popTime + 0.09);

          const popGain = this.ctx.createGain();
          popGain.gain.setValueAtTime(0.5, popTime);
          popGain.gain.exponentialRampToValueAtTime(0.001, popTime + 0.09);

          popOsc.connect(popGain);
          popGain.connect(this.ctx.destination);
          popOsc.start(popTime);
          popOsc.stop(popTime + 0.09);
        }
      }

      updateEngine(speedKmH, isNitro) {
        if (!this.isInitialized || !this.ctx || !this.isEngineRunning) return;
        const now = this.ctx.currentTime;

        if (this.wasAccelerating && !isNitro && speedKmH > 130) {
          this.playExhaustPop();
        }
        this.wasAccelerating = isNitro;

        const normSpeed = Math.max(0, Math.min(1, (speedKmH - 10) / 310));
        const baseFreq = 22 + normSpeed * 130 + (isNitro ? 35 : 0);

        this.subRumbleOsc.frequency.setTargetAtTime(baseFreq * 0.7, now, 0.04);
        this.cylOsc1.frequency.setTargetAtTime(baseFreq, now, 0.04);
        this.cylOsc2.frequency.setTargetAtTime(baseFreq * 1.25, now, 0.04);
        this.cylOsc3.frequency.setTargetAtTime(baseFreq * 1.75, now, 0.04);

        const filterFreq = 180 + Math.pow(normSpeed, 1.2) * 2600 + (isNitro ? 900 : 0);
        this.engineFilter.frequency.setTargetAtTime(filterFreq, now, 0.04);
        this.engineGain.gain.setTargetAtTime(0.14 + normSpeed * 0.10, now, 0.05);

        const noiseFilterFreq = 300 + normSpeed * 2200;
        const noiseVol = 0.015 + normSpeed * 0.07 + (isNitro ? 0.04 : 0);
        this.exhaustFilter.frequency.setTargetAtTime(noiseFilterFreq, now, 0.05);
        this.exhaustGain.gain.setTargetAtTime(noiseVol, now, 0.05);

        const turboFreq1 = 900 + Math.pow(normSpeed, 1.35) * 3200 + (isNitro ? 1100 : 0);
        const turboFreq2 = turboFreq1 * 1.03;
        const turboVol = 0.005 + normSpeed * 0.06 + (isNitro ? 0.04 : 0);

        this.turboOsc1.frequency.setTargetAtTime(turboFreq1, now, 0.05);
        this.turboOsc2.frequency.setTargetAtTime(turboFreq2, now, 0.05);
        this.turboFilter.frequency.setTargetAtTime(turboFreq1, now, 0.05);
        this.turboGain.gain.setTargetAtTime(turboVol, now, 0.05);
      }

      playCrash() {
        if (!this.isInitialized || !this.ctx) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'square';
        osc.frequency.setValueAtTime(150, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(30, this.ctx.currentTime + 0.5);

        gain.gain.setValueAtTime(0.5, this.ctx.currentTime);

        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.5);
      }

    function createRoadTexture() {
      const canvas = document.createElement('canvas');
      canvas.width = 512;
      canvas.height = 1024;
      const ctx = canvas.getContext('2d');

      ctx.fillStyle = '#181b22';
      ctx.fillRect(0, 0, 512, 1024);

      ctx.fillStyle = 'rgba(255,255,255,0.03)';
      for (let i = 0; i < 4000; i++) {
        ctx.fillRect(Math.random() * 512, Math.random() * 1024, 2, 2);
      }

      ctx.fillStyle = '#ffffff';
      ctx.fillRect(16, 0, 10, 1024);
      ctx.fillRect(512 - 26, 0, 10, 1024);

      ctx.fillStyle = '#e2e8f0';
      const laneWidth = (512 - 52) / 4;
      const dashLen = 60;
      const gapLen = 40;

      for (let l = 1; l < 4; l++) {
        const x = 26 + l * laneWidth - 3;
        for (let y = 0; y < 1024; y += dashLen + gapLen) {

    const TRAFFIC_COLORS = [0xff5500, 0x00aaee, 0xeeeeee, 0x9933ff, 0x33cc66, 0xffaa00];

    function spawnTraffic() {
      const diffCfg = DIFFICULTY_SETTINGS[selectedDifficulty];
      if (trafficCars.length >= diffCfg.trafficMax) return;

      const lane = Math.floor(Math.random() * LANE_COUNT);
      const spawnZ = 160 + Math.random() * 60;

      const occupied = trafficCars.some(c => c.lane === lane && Math.abs(c.z - spawnZ) < 14);
      if (occupied) return;

      const color = TRAFFIC_COLORS[Math.floor(Math.random() * TRAFFIC_COLORS.length)];
      const res = createSportsCarMesh(color, false);
      
      const speedOffset = (Math.random() - 0.5) * 20;
      const trafficCar = {
        lane: lane,
        x: LANE_X[lane],
        targetX: LANE_X[lane],
        z: spawnZ,
        speed: 60 + speedOffset,
        meshGroup: res.carGroup,
        wheels: res.wheels,
        laneChangeTimer: Math.random() * 5 + 3
      };

      trafficCar.meshGroup.position.set(trafficCar.x, 0, trafficCar.z);
      scene.add(trafficCar.meshGroup);
      trafficCars.push(trafficCar);
    }

    function updateTraffic(dt, playerSpeedKmH) {
      const diffCfg = DIFFICULTY_SETTINGS[selectedDifficulty];
      const playerSpeedWorld = (playerSpeedKmH * 1000) / 3600;

      for (let i = trafficCars.length - 1; i >= 0; i--) {
        const car = trafficCars[i];
        const carSpeedWorld = (car.speed * 1000) / 3600;

        car.z -= (playerSpeedWorld - carSpeedWorld) * dt;

        car.laneChangeTimer -= dt;
        if (car.laneChangeTimer <= 0) {
          car.laneChangeTimer = Math.random() * 6 + 4;
          if (Math.random() < diffCfg.laneChangeProb) {
            const side = Math.random() < 0.5 ? -1 : 1;
            const newLane = Math.max(0, Math.min(LANE_COUNT - 1, car.lane + side));
            car.lane = newLane;
            car.targetX = LANE_X[newLane];
          }
        }

        car.x += (car.targetX - car.x) * Math.min(1, dt * 6);
        car.meshGroup.position.set(car.x, 0, car.z);
        car.wheels.forEach(w => w.rotation.x += carSpeedWorld * dt * 0.8);

        if (car.z < -25 || car.z > 250) {
          scene.remove(car.meshGroup);
          trafficCars.splice(i, 1);
        }
      }
    }

    function spawnCoinsAndPowerups() {
      if (Math.random() < 0.04 && spawnedCoins.length < 15) {
        const lane = Math.floor(Math.random() * LANE_COUNT);
        const z = 140 + Math.random() * 50;
        
        const coinGeo = new THREE.CylinderGeometry(0.5, 0.5, 0.12, 16);
        coinGeo.rotateX(Math.PI / 2);
        const coinMat = new THREE.MeshStandardMaterial({ color: 0xffcc00, metalness: 0.9, roughness: 0.1, emissive: 0xaa8800, emissiveIntensity: 0.5 });
        const coinMesh = new THREE.Mesh(coinGeo, coinMat);
        coinMesh.position.set(LANE_X[lane], 0.8, z);
        scene.add(coinMesh);

        spawnedCoins.push({ mesh: coinMesh, lane, z, x: LANE_X[lane] });
      }

      if (Math.random() < 0.008 && spawnedPowerups.length < 2) {
        const lane = Math.floor(Math.random() * LANE_COUNT);
        const z = 160 + Math.random() * 40;
        const types = ['magnet', 'shield', 'double'];
        const type = types[Math.floor(Math.random() * types.length)];

        let geo, mat;
        if (type === 'magnet') {
          geo = new THREE.TorusGeometry(0.5, 0.2, 8, 16);
          mat = new THREE.MeshStandardMaterial({ color: 0xff2244, emissive: 0xff0022, emissiveIntensity: 0.8 });
        } else if (type === 'shield') {
          geo = new THREE.IcosahedronGeometry(0.6, 0);
          mat = new THREE.MeshStandardMaterial({ color: 0x00ffff, emissive: 0x0088ff, emissiveIntensity: 0.8 });
        } else {
          geo = new THREE.OctahedronGeometry(0.65, 0);
          mat = new THREE.MeshStandardMaterial({ color: 0x00ff66, emissive: 0x00cc44, emissiveIntensity: 0.8 });
        }

        const mesh = new THREE.Mesh(geo, mat);
        mesh.position.set(LANE_X[lane], 1.0, z);
        scene.add(mesh);

        spawnedPowerups.push({ mesh, type, lane, z, x: LANE_X[lane] });
      }
    }

    function updateCoinsAndPowerups(dt, playerSpeedKmH) {
      const speedWorld = (playerSpeedKmH * 1000) / 3600;

      for (let i = spawnedCoins.length - 1; i >= 0; i--) {
        const c = spawnedCoins[i];
        
        if (activePowerups.magnetTimer > 0) {
          const distToPlayer = Math.hypot(c.x - player.x, c.z - 0);
          if (distToPlayer < 22) {
            c.x += (player.x - c.x) * Math.min(1, dt * 10);
            c.z += (0 - c.z) * Math.min(1, dt * 10);
          } else {
            c.z -= speedWorld * dt;
          }
        } else {
          c.z -= speedWorld * dt;
        }

        c.mesh.position.set(c.x, 0.8, c.z);
        c.mesh.rotation.z += dt * 4.0;

        if (Math.abs(c.z) < 1.8 && Math.abs(c.x - player.x) < 1.4) {
          coins += 1;
          score += 50 * (activePowerups.doubleTimer > 0 ? 2 : 1);
          soundEngine.playCoin();
          createFloatingText('+1 MOEDA 🪙', 'text-yellow-300 font-bold');
          document.getElementById('coins-text').innerText = coins;

          scene.remove(c.mesh);
          spawnedCoins.splice(i, 1);
          continue;
        }

        if (c.z < -20) {
          scene.remove(c.mesh);
          spawnedCoins.splice(i, 1);
        }
      }

      for (let i = spawnedPowerups.length - 1; i >= 0; i--) {
        const p = spawnedPowerups[i];
        p.z -= speedWorld * dt;
        p.mesh.position.set(p.x, 1.0, p.z);
        p.mesh.rotation.y += dt * 3.0;

        if (Math.abs(p.z) < 1.8 && Math.abs(p.x - player.x) < 1.4) {
          if (p.type === 'magnet') {
            activePowerups.magnetTimer = 10.0;
            createFloatingText('🧲 ÍMÃ DE MOEDAS!', 'text-red-400 font-black');
          } else if (p.type === 'shield') {
            activePowerups.shieldActive = true;
            if (playerShieldMesh) playerShieldMesh.visible = true;
            createFloatingText('🛡️ ESCUDO DE ENERGIA!', 'text-cyan-400 font-black');
          } else if (p.type === 'double') {
            activePowerups.doubleTimer = 10.0;
            createFloatingText('⚡ 2X PONTOS!', 'text-emerald-400 font-black');
          }
          soundEngine.playPowerup();
          scene.remove(p.mesh);
          spawnedPowerups.splice(i, 1);
          continue;
        }

        if (p.z < -20) {
          scene.remove(p.mesh);
          spawnedPowerups.splice(i, 1);
        }
      }
    }

    let barricadeStripeTexture = null;
    function getBarricadeStripeTexture() {
      if (barricadeStripeTexture) return barricadeStripeTexture;
      const canvas = document.createElement('canvas');
      canvas.width = 256;
      canvas.height = 64;
      const ctx = canvas.getContext('2d');
      ctx.fillStyle = '#ff5500';
      ctx.fillRect(0, 0, 256, 64);
      ctx.fillStyle = '#ffffff';
      ctx.beginPath();
      for (let x = -64; x < 320; x += 32) {
        ctx.moveTo(x, 0);
        ctx.lineTo(x + 20, 0);
        ctx.lineTo(x - 12, 64);
        ctx.lineTo(x - 32, 64);
        ctx.closePath();
      }
      ctx.fill();
      barricadeStripeTexture = new THREE.CanvasTexture(canvas);
      barricadeStripeTexture.wrapS = THREE.RepeatWrapping;
      barricadeStripeTexture.wrapT = THREE.ClampToEdgeWrapping;
      return barricadeStripeTexture;
    }

    function createConstructionBarricade() {
      const group = new THREE.Group();

      const legMat = new THREE.MeshStandardMaterial({ color: 0x1e293b, metalness: 0.8, roughness: 0.3 });
      const legGeo = new THREE.BoxGeometry(0.35, 0.12, 1.2);
      const leftFoot = new THREE.Mesh(legGeo, legMat);
      leftFoot.position.set(-1.1, 0.06, 0);
      const rightFoot = new THREE.Mesh(legGeo, legMat);
      rightFoot.position.set(1.1, 0.06, 0);
      group.add(leftFoot, rightFoot);

      const postGeo = new THREE.BoxGeometry(0.12, 1.3, 0.12);
      const leftPost = new THREE.Mesh(postGeo, legMat);
      leftPost.position.set(-1.1, 0.65, 0);
      const rightPost = new THREE.Mesh(postGeo, legMat);
      rightPost.position.set(1.1, 0.65, 0);
      group.add(leftPost, rightPost);

      const plankMat = new THREE.MeshStandardMaterial({
        map: getBarricadeStripeTexture(),
        roughness: 0.4,
        metalness: 0.1
      });
      const plankGeo = new THREE.BoxGeometry(2.4, 0.35, 0.08);
      const plankTop = new THREE.Mesh(plankGeo, plankMat);
      plankTop.position.set(0, 1.05, 0);
      const plankBottom = new THREE.Mesh(plankGeo, plankMat);
      plankBottom.position.set(0, 0.52, 0);
      group.add(plankTop, plankBottom);

      const beaconMat = new THREE.MeshStandardMaterial({
        color: 0xff9900,
        emissive: 0xffaa00,
        emissiveIntensity: 2.2,
        roughness: 0.2
      });
      const beaconGeo = new THREE.CylinderGeometry(0.13, 0.13, 0.28, 12);
      const beaconL = new THREE.Mesh(beaconGeo, beaconMat);
      beaconL.position.set(-1.1, 1.44, 0);
      const beaconR = new THREE.Mesh(beaconGeo, beaconMat);
      beaconR.position.set(1.1, 1.44, 0);
      group.add(beaconL, beaconR);

      const coneBaseGeo = new THREE.BoxGeometry(0.55, 0.06, 0.55);
      const coneBaseMat = new THREE.MeshStandardMaterial({ color: 0x0f172a, roughness: 0.9 });
      const coneBase = new THREE.Mesh(coneBaseGeo, coneBaseMat);
      coneBase.position.set(-1.45, 0.03, 0.35);

      const coneBodyGeo = new THREE.ConeGeometry(0.24, 0.85, 14);
      const coneBodyMat = new THREE.MeshStandardMaterial({ color: 0xff4400, roughness: 0.3 });
      const coneBody = new THREE.Mesh(coneBodyGeo, coneBodyMat);
      coneBody.position.set(-1.45, 0.45, 0.35);

      const stripeGeo = new THREE.CylinderGeometry(0.15, 0.19, 0.2, 14);
      const stripeMat = new THREE.MeshStandardMaterial({ color: 0xf8fafc, roughness: 0.2 });
      const stripe = new THREE.Mesh(stripeGeo, stripeMat);
      stripe.position.set(-1.45, 0.45, 0.35);

      group.add(coneBase, coneBody, stripe);
      return group;
    }

    function spawnHazards() {
      if (Math.random() < 0.015 && spawnedHazards.length < 4) {
        const lane = Math.floor(Math.random() * LANE_COUNT);
        const z = 150 + Math.random() * 50;
        const isOil = Math.random() < 0.6;

        let mesh;
        if (isOil) {
          const oilGeo = new THREE.CircleGeometry(1.4, 16);
          oilGeo.rotateX(-Math.PI / 2);
          const oilMat = new THREE.MeshStandardMaterial({ color: 0x05070a, metalness: 0.9, roughness: 0.1, transparent: true, opacity: 0.9 });
          mesh = new THREE.Mesh(oilGeo, oilMat);
          mesh.position.set(LANE_X[lane], 0.02, z);
        } else {
          mesh = createConstructionBarricade();
          mesh.position.set(LANE_X[lane], 0, z);
        }

        scene.add(mesh);
        spawnedHazards.push({ mesh, isOil, lane, z, x: LANE_X[lane] });
      }
    }

    function updateHazards(dt, playerSpeedKmH) {
      const speedWorld = (playerSpeedKmH * 1000) / 3600;

      for (let i = spawnedHazards.length - 1; i >= 0; i--) {
        const h = spawnedHazards[i];
        h.z -= speedWorld * dt;
        h.mesh.position.z = h.z;

        if (Math.abs(h.z) < 2.0 && Math.abs(h.x - player.x) < 1.3) {
          if (h.isOil) {
            if (oilSpinTimer <= 0) {
              oilSpinTimer = 1.0;
              soundEngine.playSlick();
              createFloatingText('⚠️ DERROPA NA MANCHA DE ÓLEO!', 'text-yellow-400 font-black');
            }
          } else {
            if (activePowerups.shieldActive) {
              activePowerups.shieldActive = false;
              if (playerShieldMesh) playerShieldMesh.visible = false;
              soundEngine.playShieldBreak();
              createFloatingText('🛡️ ESCUDO ABSORVEU O IMPACTO!', 'text-cyan-300 font-bold');
            } else {
              triggerGameOver('Colisão com Barricada de Obras!');
              return;
            }
          }
          scene.remove(h.mesh);
          spawnedHazards.splice(i, 1);
          continue;
        }

        if (h.z < -20) {
          scene.remove(h.mesh);
          spawnedHazards.splice(i, 1);
        }
      }
    }

    function checkCollisions() {
      for (let i = trafficCars.length - 1; i >= 0; i--) {
        const car = trafficCars[i];
        if (Math.abs(car.z) < 3.2 && Math.abs(car.x - player.x) < 1.65) {
          if (activePowerups.shieldActive) {
            activePowerups.shieldActive = false;
            if (playerShieldMesh) playerShieldMesh.visible = false;
            soundEngine.playShieldBreak();
            createFloatingText('🛡️ ESCUDO DESTRUIU O VEÍCULO!', 'text-cyan-300 font-black');
            scene.remove(car.meshGroup);
            trafficCars.splice(i, 1);
          } else {
            triggerGameOver('Colisão com Veículo no Tráfego!');
            return;
          }
        }
      }
    }

    function triggerGameOver(reasonText) {

      localStorage.setItem('car-3d-highscore', String(highScore));

      document.getElementById('gameover-reason').innerText = reasonText;
      document.getElementById('final-score').innerText = Math.floor(score);
      document.getElementById('final-highscore').innerText = highScore;
      document.getElementById('gameover-screen').classList.remove('hidden');
    }

    function animate() {
      requestAnimationFrame(animate);

      const dt = Math.min(clock ? clock.getDelta() : 0.016, 0.1);
      const diffCfg = DIFFICULTY_SETTINGS[selectedDifficulty];

      if (running) {
        if (activePowerups.magnetTimer > 0) activePowerups.magnetTimer = Math.max(0, activePowerups.magnetTimer - dt);
        if (activePowerups.doubleTimer > 0) activePowerups.doubleTimer = Math.max(0, activePowerups.doubleTimer - dt);
        updatePowerupHUD();

        let targetSpeed = baseSpeed;
        if (keys.up) targetSpeed = baseSpeed + 120;
        else if (keys.down) targetSpeed = 10;

        speedKmH += (targetSpeed - speedKmH) * Math.min(1, dt * 3.5);
        speedKmH = Math.max(10, speedKmH);
        baseSpeed += dt * diffCfg.speedAccel;

        const speedWorld = (speedKmH * 1000) / 3600;
        distance += speedWorld * dt;
        score += speedWorld * dt * 0.5 * (activePowerups.doubleTimer > 0 ? 2.0 : 1.0) * diffCfg.scoreMult;

        if (oilSpinTimer > 0) {
          oilSpinTimer -= dt;
          player.meshGroup.rotation.y += dt * 18.0;
        } else {
          const dx = player.targetX - player.x;
          player.x += dx * Math.min(1, player.lerpSpeed * dt);
          player.meshGroup.position.x = player.x;
          player.meshGroup.rotation.y = dx * 0.08;
        }

        player.wheels.forEach(w => w.rotation.x += speedWorld * dt * 0.8);
        player.nitroFlames.forEach(f => f.visible = keys.up || speedKmH > 220);

        roadTexture.offset.y -= (speedWorld * dt) / 20;
        roadsideProps.forEach(prop => {
          prop.position.z -= speedWorld * dt;
          if (prop.position.z < -30) prop.position.z += 360;
        });

        biomeSwitchTimer += dt;
        if (biomeSwitchTimer > 20.0) {
          biomeSwitchTimer = 0;
          applyBiome(currentBiomeIdx + 1);
        }

        spawnTraffic();
        updateTraffic(dt, speedKmH);

        spawnCoinsAndPowerups();
        updateCoinsAndPowerups(dt, speedKmH);

        spawnHazards();
        updateHazards(dt, speedKmH);

        checkCollisions();

        soundEngine.updateEngine(speedKmH, keys.up);

        document.getElementById('score-text').innerText = Math.floor(score);
        document.getElementById('highscore-text').innerText = highScore;
      }

      if (renderer && scene && camera) {
        renderer.render(scene, camera);
      }
      drawSpeedometerCanvas(speedKmH);
      drawMinimapCanvas();
    }

    const minimapCanvas = document.getElementById('minimap');
    const minimapCtx = minimapCanvas.getContext('2d');

    function drawMinimapCanvas() {
      const w = minimapCanvas.width, h = minimapCanvas.height;
      minimapCtx.clearRect(0, 0, w, h);

      minimapCtx.strokeStyle = 'rgba(0, 255, 255, 0.2)';
      minimapCtx.lineWidth = 1;
      for (let l = 0; l <= LANE_COUNT; l++) {
        const x = (w / LANE_COUNT) * l;
        minimapCtx.beginPath();
        minimapCtx.moveTo(x, 0);
        minimapCtx.lineTo(x, h);
        minimapCtx.stroke();
      }

      const mapXToCanvas = (x) => ((x + 8) / 16) * w;
      const px = mapXToCanvas(player.x);
      const py = h - 15;

      minimapCtx.fillStyle = '#00ffff';
      minimapCtx.beginPath();
      minimapCtx.arc(px, py, 4, 0, Math.PI * 2);
      minimapCtx.fill();

      minimapCtx.fillStyle = '#ff2a5f';
      trafficCars.forEach(car => {
        if (car.z > -10 && car.z < 160) {
          const tx = mapXToCanvas(car.x);
          const ty = h - ((car.z + 10) / 170) * h;
          minimapCtx.fillRect(tx - 2, ty - 2, 4, 4);
        }
      });
    }

    function onWindowResize() {
      if (!camera || !renderer) return;
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
      checkOrientation();
    }

    /* =========================================================================
       MOBILE DETECTOR & ORIENTATION LOCK CHECK
       ========================================================================= */
    let overrideOrientation = false;

    function dismissRotateOverlay() {
      overrideOrientation = true;
      const overlay = document.getElementById('rotate-device-overlay');
      if (overlay) {
        overlay.classList.add('hidden');
        overlay.style.display = 'none';
      }
    }

    function isMobileDevice() {
      const ua = navigator.userAgent || navigator.vendor || window.opera;
      const isMobileUA = /Android|webOS|iPhone|iPod|BlackBerry|IEMobile|Opera Mini|Mobile|mobile|CriOS|FxiOS/i.test(ua);
      const isTabletUA = /iPad|Tablet/i.test(ua) || (navigator.platform === 'MacIntel' && navigator.maxTouchPoints > 1);
      return isMobileUA || isTabletUA;
    }

    function checkOrientation() {
      const overlay = document.getElementById('rotate-device-overlay');
      const introEl = document.getElementById('intro-screen');
      const startEl = document.getElementById('start-screen');
      if (!overlay) return;

      if (overrideOrientation || running) {
        overlay.classList.add('hidden');
        overlay.style.display = 'none';
        return;
      }

      const isMobile = isMobileDevice();
      const isSmallPortrait = isMobile && (window.innerHeight > window.innerWidth) && (window.innerWidth < 600);

      if (isSmallPortrait) {
        overlay.classList.remove('hidden');
        overlay.style.display = 'flex';
        if (introEl) {
          introEl.classList.add('hidden');
          introEl.style.display = 'none';
        }
        if (startEl) startEl.classList.add('hidden');
      } else {
        overlay.classList.add('hidden');
        overlay.style.display = 'none';
        if (!introPlayed && introEl) {
          introEl.classList.remove('hidden');
          introEl.style.display = 'flex';
          if (startEl) startEl.classList.add('hidden');
        } else if (startEl && !running && (!introEl || introEl.classList.contains('hidden'))) {
          startEl.classList.remove('hidden');
        }
      }
    }

    window.addEventListener('resize', checkOrientation);
    window.addEventListener('orientationchange', checkOrientation);

    // Keyboard Input
    window.addEventListener('keydown', e => {
      soundEngine.init();
      const introEl = document.getElementById('intro-screen');
      if (!introPlayed && introEl && !introEl.classList.contains('hidden')) {
        if (e.code === 'Space' || e.code === 'Enter') {
          e.preventDefault();
          dismissIntro();
          return;
        }
      }
      if (e.code === 'ArrowLeft' || e.code === 'KeyA') moveLeft();
      else if (e.code === 'ArrowRight' || e.code === 'KeyD') moveRight();
      else if (e.code === 'ArrowUp' || e.code === 'KeyW') keys.up = true;
      else if (e.code === 'ArrowDown' || e.code === 'KeyS') keys.down = true;
      else if (e.code === 'Space' || e.code === 'Enter') {
        e.preventDefault();
        startGame();
      }
    });

    window.addEventListener('keyup', e => {
      if (e.code === 'ArrowUp' || e.code === 'KeyW') keys.up = false;
      else if (e.code === 'ArrowDown' || e.code === 'KeyS') keys.down = false;
    });

    // Touch Control Binding
    function setupTouchControlEvents() {
      const bindControlBtn = (id, keyName) => {
        const el = document.getElementById(id);
        if (!el) return;
        const press = (e) => {
          soundEngine.init();
          if (!running) return;
          e.preventDefault();
          keys[keyName] = true;
        };
        const release = (e) => {
          if (!running) return;
          e.preventDefault();
          keys[keyName] = false;
        };
        el.addEventListener('pointerdown', press);
        el.addEventListener('pointerup', release);
        el.addEventListener('pointerleave', release);
        el.addEventListener('touchstart', press, { passive: false });
        el.addEventListener('touchend', release, { passive: false });
      };

      bindControlBtn('btn-nitro', 'up');
      bindControlBtn('btn-brake', 'down');

      const btnLeft = document.getElementById('btn-left');
      if (btnLeft) {
        const triggerLeft = (e) => {
          soundEngine.init();
          if (!running) return;
          e.preventDefault();
          moveLeft();
        };
        btnLeft.addEventListener('pointerdown', triggerLeft);
        btnLeft.addEventListener('touchstart', triggerLeft, { passive: false });
      }

      const btnRight = document.getElementById('btn-right');
      if (btnRight) {
        const triggerRight = (e) => {
          soundEngine.init();
          if (!running) return;
          e.preventDefault();
          moveRight();
        };
        btnRight.addEventListener('pointerdown', triggerRight);
        btnRight.addEventListener('touchstart', triggerRight, { passive: false });
      }
    }

    window.onload = () => {
      initThree();
      checkOrientation();

      const introEl = document.getElementById('intro-screen');
      if (introEl) {
        introEl.addEventListener('pointerdown', (e) => {
          e.preventDefault();
          dismissIntro();
        });
      }

      const bindUIStart = (btnId) => {
        const btn = document.getElementById(btnId);
        if (btn) {
          const handler = (e) => {
            e.preventDefault();
            e.stopPropagation();
            startGame();
          };
          btn.addEventListener('pointerdown', handler);
          btn.addEventListener('click', handler);
        }
      };

      bindUIStart('start-btn');
      bindUIStart('restart-btn');

      setupTouchControlEvents();
      loadControlsConfig();
      animate();
    };

    /* =========================================================================
       CONTROLS CUSTOMIZER LOGIC
       ========================================================================= */
    let touchControlsConfig = {
      steeringPos: { x: 16, y: 24, alignRight: false },
      speedPos: { x: 16, y: 24, alignRight: true },
      scale: 1.0
    };

    function loadControlsConfig() {
      const saved = localStorage.getItem('rr3-touch-config');
      if (saved) {
        try { touchControlsConfig = JSON.parse(saved); } catch(e) {}
      }
      applyControlsConfig();
      setupControlsCustomizer();
    }

    function setupControlsCustomizer() {
      const slider = document.getElementById('controls-scale-slider');
      if (slider) {
        slider.addEventListener('input', (e) => {
          touchControlsConfig.scale = parseFloat(e.target.value);
          applyControlsConfig();
        });
      }

      const makeDraggable = (el, posKey) => {
        let isDragging = false;
        let startX, startY, initialPos;

        const onStart = (e) => {
          const customizer = document.getElementById('customizer-screen');
          if (!customizer || customizer.classList.contains('hidden')) return;
          isDragging = true;
          const touch = e.touches ? e.touches[0] : e;
          startX = touch.clientX;
          startY = touch.clientY;
          initialPos = { ...touchControlsConfig[posKey] };
        };

        const onMove = (e) => {
          if (!isDragging) return;
          const touch = e.touches ? e.touches[0] : e;
          const dx = touch.clientX - startX;
          const dy = touch.clientY - startY;

          if (touchControlsConfig[posKey].alignRight) {
            touchControlsConfig[posKey].x = Math.max(0, initialPos.x - dx);
          } else {
            touchControlsConfig[posKey].x = Math.max(0, initialPos.x + dx);
          }
          touchControlsConfig[posKey].y = Math.max(0, initialPos.y - dy);
          applyControlsConfig();
        };

        const onEnd = () => { isDragging = false; };

        el.addEventListener('mousedown', onStart);
        el.addEventListener('touchstart', onStart, { passive: true });
        window.addEventListener('mousemove', onMove);
        window.addEventListener('touchmove', onMove, { passive: true });
        window.addEventListener('mouseup', onEnd);
        window.addEventListener('touchend', onEnd);
      };

      const steeringEl = document.getElementById('steering-group');
      const speedEl = document.getElementById('speed-group');
      if (steeringEl) makeDraggable(steeringEl, 'steeringPos');
      if (speedEl) makeDraggable(speedEl, 'speedPos');
    }

    function applyControlsConfig() {
      const steeringEl = document.getElementById('steering-group');
      const speedEl = document.getElementById('speed-group');
      const scaleSlider = document.getElementById('controls-scale-slider');
      const scaleText = document.getElementById('scale-val-text');

      if (steeringEl) {
        steeringEl.style.left = touchControlsConfig.steeringPos.alignRight ? 'auto' : `${touchControlsConfig.steeringPos.x}px`;
        steeringEl.style.right = touchControlsConfig.steeringPos.alignRight ? `${touchControlsConfig.steeringPos.x}px` : 'auto';
        steeringEl.style.bottom = `${touchControlsConfig.steeringPos.y}px`;
        steeringEl.style.transform = `scale(${touchControlsConfig.scale})`;
      }

      if (speedEl) {
        speedEl.style.left = touchControlsConfig.speedPos.alignRight ? 'auto' : `${touchControlsConfig.speedPos.x}px`;
        speedEl.style.right = touchControlsConfig.speedPos.alignRight ? `${touchControlsConfig.speedPos.x}px` : 'auto';
        speedEl.style.bottom = `${touchControlsConfig.speedPos.y}px`;
        speedEl.style.transform = `scale(${touchControlsConfig.scale})`;
      }

      if (scaleSlider) scaleSlider.value = touchControlsConfig.scale;
      if (scaleText) scaleText.innerText = `${Math.round(touchControlsConfig.scale * 100)}%`;
    }

    function openControlsCustomizer() {
      document.getElementById('start-screen').classList.add('hidden');
      document.getElementById('touch-controls').classList.remove('hidden');
      document.getElementById('customizer-screen').classList.remove('hidden');
    }

    function closeControlsCustomizer() {
      localStorage.setItem('rr3-touch-config', JSON.stringify(touchControlsConfig));
      document.getElementById('customizer-screen').classList.add('hidden');
      document.getElementById('start-screen').classList.remove('hidden');
    }

    function resetControlsConfig() {
      touchControlsConfig = { steeringPos: { x: 16, y: 24, alignRight: false }, speedPos: { x: 16, y: 24, alignRight: true }, scale: 1.0 };
      applyControlsConfig();
      localStorage.setItem('rr3-touch-config', JSON.stringify(touchControlsConfig));
    }
